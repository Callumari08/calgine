47// ---- Collision layers ----
48// A Jolt ObjectLayer packs the user's collision layer and whether the body can move:
49// object_layer = (collision_layer << 1) | moving
50// Static bodies go in the "non_moving" broad phase tree and never test against each other.
51
52namespace PhysicsBroadPhaseLayers
53{
54 static constexpr JPH::BroadPhaseLayer non_moving(0);
55 static constexpr JPH::BroadPhaseLayer moving(1);
56 static constexpr JPH::uint count = 2;
57}
58
59inline JPH::ObjectLayer make_object_layer(CollisionLayer layer, bool moving)
60{
61 return static_cast<JPH::ObjectLayer>((static_cast<uint32_t>(layer) << 1) | (moving ? 1u : 0u));
62}
63inline CollisionLayer collision_layer_of(JPH::ObjectLayer object_layer) { return static_cast<CollisionLayer>(object_layer >> 1); }
64inline bool is_moving_layer(JPH::ObjectLayer object_layer) { return (object_layer & 1u) != 0; }

109// Jolt calls these from its worker threads during PhysicsSystem::Update(), so we only record body IDs

112struct QueuedContact

114 bool is_enter = true;
115 JPH::BodyID body_1;

118 glm::vec3 normal = glm::vec3(0.0f);

121class ContactQueue final : public JPH::ContactListener
122{
123public:
124 void OnContactAdded(const JPH::Body& body_1, const JPH::Body& body_2, const JPH::ContactManifold& manifold, JPH::ContactSettings&) override
125 {
126 QueuedContact contact;
127 contact.is_enter = true;
128 contact.body_1 = body_1.GetID();
129 contact.body_2 = body_2.GetID();
130 if (manifold.mRelativeContactPointsOn1.size() > 0)
131 contact.point = to_glm(manifold.GetWorldSpaceContactPointOn1(0));
132 contact.normal = to_glm(manifold.mWorldSpaceNormal);
133
134 std::lock_guard<std::mutex> lock(mutex);
135 contacts.push_back(contact);
136 }
137
138 void OnContactRemoved(const JPH::SubShapeIDPair& pair) override
139 {
140 QueuedContact contact;
141 contact.is_enter = false;
142 contact.body_1 = pair.GetBody1ID();
143 contact.body_2 = pair.GetBody2ID();
144
145 std::lock_guard<std::mutex> lock(mutex);
146 contacts.push_back(contact);
147 }
148
149 std::vector<QueuedContact> take()
150 {
151 std::lock_guard<std::mutex> lock(mutex);
152 std::vector<QueuedContact> out;
153 out.swap(contacts);
154 return out;
155 }
156
157private:
158 std::mutex mutex;
159 std::vector<QueuedContact> contacts;
160};
