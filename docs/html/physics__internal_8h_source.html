101// ---- Contact queue ----
102// Jolt calls these from its worker threads during PhysicsSystem::Update(), so we only record body IDs
103// here. PhysicsWorld::step() turns them into CollisionEvents on the main thread afterwards.
104
105struct QueuedContact
106{
107 bool is_enter = true;
108 JPH::BodyID body_1;
109 JPH::BodyID body_2;
110 glm::vec3 point = glm::vec3(0.0f);
111 glm::vec3 normal = glm::vec3(0.0f);
112};

114class ContactQueue final : public JPH::ContactListener
115{
116public:
117 void OnContactAdded(const JPH::Body& body_1, const JPH::Body& body_2, const JPH::ContactManifold& manifold, JPH::ContactSettings&) override
118 {
119 QueuedContact contact;
120 contact.is_enter = true;
121 contact.body_1 = body_1.GetID();
122 contact.body_2 = body_2.GetID();
123 if (manifold.mRelativeContactPointsOn1.size() > 0)
124 contact.point = to_glm(manifold.GetWorldSpaceContactPointOn1(0));
125 contact.normal = to_glm(manifold.mWorldSpaceNormal);
126
127 std::lock_guard<std::mutex> lock(mutex);
128 contacts.push_back(contact);
129 }
130
131 void OnContactRemoved(const JPH::SubShapeIDPair& pair) override
132 {
133 QueuedContact contact;
134 contact.is_enter = false;
135 contact.body_1 = pair.GetBody1ID();
136 contact.body_2 = pair.GetBody2ID();
137
138 std::lock_guard<std::mutex> lock(mutex);
139 contacts.push_back(contact);
140 }
141
142 std::vector<QueuedContact> take()
143 {
144 std::lock_guard<std::mutex> lock(mutex);
145 std::vector<QueuedContact> out;
146 out.swap(contacts);
147 return out;
148 }
149
150private:
151 std::mutex mutex;
152 std::vector<QueuedContact> contacts;
153};
