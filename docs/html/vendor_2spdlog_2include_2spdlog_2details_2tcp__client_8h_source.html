43 int connect_socket_with_timeout(int sockfd,
44 const struct sockaddr *addr,
45 socklen_t addrlen,
46 const timeval &tv) {
47 // Blocking connect if timeout is zero
48 if (tv.tv_sec == 0 && tv.tv_usec == 0) {
49 int rv = ::connect(sockfd, addr, addrlen);
50 if (rv < 0 && errno == EISCONN) {
51 // already connected, treat as success
52 return 0;
53 }
54 return rv;
55 }
56
57 // Non-blocking path
58 int orig_flags = ::fcntl(sockfd, F_GETFL, 0);
59 if (orig_flags < 0) {
60 return -1;
61 }
62 if (::fcntl(sockfd, F_SETFL, orig_flags | O_NONBLOCK) < 0) {
63 return -1;
64 }
65
66 int rv = ::connect(sockfd, addr, addrlen);
67 if (rv == 0 || (rv < 0 && errno == EISCONN)) {
68 // immediate connect or already connected
69 ::fcntl(sockfd, F_SETFL, orig_flags);
70 return 0;
71 }
72 if (errno != EINPROGRESS) {
73 ::fcntl(sockfd, F_SETFL, orig_flags);
74 return -1;
75 }
76
77 // wait for writability
78 fd_set wfds;
79 FD_ZERO(&wfds);
80 FD_SET(sockfd, &wfds);
81
82 struct timeval tv_copy = tv;
83 rv = ::select(sockfd + 1, nullptr, &wfds, nullptr, &tv_copy);
84 if (rv <= 0) {
85 // timeout or error
86 ::fcntl(sockfd, F_SETFL, orig_flags);
87 if (rv == 0) errno = ETIMEDOUT;
88 return -1;
89 }
90
91 // check socket error
92 int so_error = 0;
93 socklen_t len = sizeof(so_error);
94 if (::getsockopt(sockfd, SOL_SOCKET, SO_ERROR, &so_error, &len) < 0) {
95 ::fcntl(sockfd, F_SETFL, orig_flags);
96 return -1;
97 }
98 ::fcntl(sockfd, F_SETFL, orig_flags);
99 if (so_error != 0 && so_error != EISCONN) {
100 errno = so_error;
101 return -1;
102 }
103
104 return 0;
105 }

107 // try to connect or throw on failure
108 void connect(const std::string &host, int port, int timeout_ms = 0) {
109 close();
110 struct addrinfo hints {};
111 memset(&hints, 0, sizeof(struct addrinfo));
112 hints.ai_family = AF_UNSPEC; // To work with IPv4, IPv6, and so on
113 hints.ai_socktype = SOCK_STREAM; // TCP
114 hints.ai_flags = AI_NUMERICSERV; // port passed as as numeric value
115 hints.ai_protocol = 0;
116
117 struct timeval tv;
118 tv.tv_sec = timeout_ms / 1000;
119 tv.tv_usec = (timeout_ms % 1000) * 1000;
120
121 auto port_str = std::to_string(port);
122 struct addrinfo *addrinfo_result;
123 auto rv = ::getaddrinfo(host.c_str(), port_str.c_str(), &hints, &addrinfo_result);
124 if (rv != 0) {
125 throw_spdlog_ex(fmt_lib::format("::getaddrinfo failed: {}", gai_strerror(rv)));
126 }
127
128 // Try each address until we successfully connect(2).
129 int last_errno = 0;
130 for (auto *rp = addrinfo_result; rp != nullptr; rp = rp->ai_next) {
131#if defined(SOCK_CLOEXEC)
132 const int flags = SOCK_CLOEXEC;
133#else
134 const int flags = 0;
135#endif
136 socket_ = ::socket(rp->ai_family, rp->ai_socktype | flags, rp->ai_protocol);
137 if (socket_ == -1) {
138 last_errno = errno;
139 continue;
140 }
141 ::fcntl(socket_, F_SETFD, FD_CLOEXEC);
142 if (connect_socket_with_timeout(socket_, rp->ai_addr, rp->ai_addrlen, tv) == 0) {
143 last_errno = 0;
144 break;
145 }
146 last_errno = errno;
147 ::close(socket_);
148 socket_ = -1;
149 }
150 ::freeaddrinfo(addrinfo_result);
151 if (socket_ == -1) {
152 throw_spdlog_ex("::connect failed", last_errno);
153 }
154
155 if (timeout_ms > 0) {
156 // Set timeouts for send and recv
157 ::setsockopt(socket_, SOL_SOCKET, SO_RCVTIMEO, (const char *)&tv, sizeof(tv));
158 ::setsockopt(socket_, SOL_SOCKET, SO_SNDTIMEO, (const char *)&tv, sizeof(tv));
159 }
160
161 // set TCP_NODELAY
162 int enable_flag = 1;
163 ::setsockopt(socket_, IPPROTO_TCP, TCP_NODELAY, reinterpret_cast<char *>(&enable_flag),
164 sizeof(enable_flag));
165
166 // prevent sigpipe on systems where MSG_NOSIGNAL is not available
167#if defined(SO_NOSIGPIPE) && !defined(MSG_NOSIGNAL)
168 ::setsockopt(socket_, SOL_SOCKET, SO_NOSIGPIPE, reinterpret_cast<char *>(&enable_flag),
169 sizeof(enable_flag));
170#endif
171
172#if !defined(SO_NOSIGPIPE) && !defined(MSG_NOSIGNAL)
173#error "tcp_sink would raise SIGPIPE since neither SO_NOSIGPIPE nor MSG_NOSIGNAL are available"
174#endif
175 }
176
177 // Send exactly n_bytes of the given data.
178 // On error close the connection and throw.
179 void send(const char *data, size_t n_bytes) {
180 size_t bytes_sent = 0;
181 while (bytes_sent < n_bytes) {
182#if defined(MSG_NOSIGNAL)
183 const int send_flags = MSG_NOSIGNAL;
184#else
185 const int send_flags = 0;
186#endif
187 auto write_result =
188 ::send(socket_, data + bytes_sent, n_bytes - bytes_sent, send_flags);
189 if (write_result < 0) {
190 close();
191 throw_spdlog_ex("write(2) failed", errno);
192 }
193
194 if (write_result == 0) // (probably should not happen but in any case..)
195 {
196 break;
197 }
198 bytes_sent += static_cast<size_t>(write_result);
199 }
200 }
