61 int connect_socket_with_timeout(SOCKET sockfd,
62 const struct sockaddr *addr,
63 int addrlen,
64 const timeval &tv) {
65 // If no timeout requested, do a normal blocking connect.
66 if (tv.tv_sec == 0 && tv.tv_usec == 0) {
67 int rv = ::connect(sockfd, addr, addrlen);
68 if (rv == SOCKET_ERROR && WSAGetLastError() == WSAEISCONN) {
69 return 0;
70 }
71 return rv;
72 }
73
74 // Switch to non‐blocking mode
75 u_long mode = 1UL;
76 if (::ioctlsocket(sockfd, FIONBIO, &mode) == SOCKET_ERROR) {
77 return SOCKET_ERROR;
78 }
79
80 int rv = ::connect(sockfd, addr, addrlen);
81 int last_error = WSAGetLastError();
82 if (rv == 0 || last_error == WSAEISCONN) {
83 mode = 0UL;
84 if (::ioctlsocket(sockfd, FIONBIO, &mode) == SOCKET_ERROR) {
85 return SOCKET_ERROR;
86 }
87 return 0;
88 }
89 if (last_error != WSAEWOULDBLOCK) {
90 // Real error
91 mode = 0UL;
92 if (::ioctlsocket(sockfd, FIONBIO, &mode)) {
93 return SOCKET_ERROR;
94 }
95 return SOCKET_ERROR;
96 }
97
98 // Wait until socket is writable or timeout expires
99 fd_set wfds;
100 FD_ZERO(&wfds);
101 FD_SET(sockfd, &wfds);
102
103 rv = ::select(0, nullptr, &wfds, nullptr, const_cast<timeval *>(&tv));
104
105 // Restore blocking mode regardless of select result
106 mode = 0UL;
107 if (::ioctlsocket(sockfd, FIONBIO, &mode) == SOCKET_ERROR) {
108 return SOCKET_ERROR;
109 }
110
111 if (rv == 0) {
112 WSASetLastError(WSAETIMEDOUT);
113 return SOCKET_ERROR;
114 }
115 if (rv == SOCKET_ERROR) {
116 return SOCKET_ERROR;
117 }
118
119 int so_error = 0;
120 int len = sizeof(so_error);
121 if (::getsockopt(sockfd, SOL_SOCKET, SO_ERROR, reinterpret_cast<char *>(&so_error), &len) ==
122 SOCKET_ERROR) {
123 return SOCKET_ERROR;
124 }
125 if (so_error != 0 && so_error != WSAEISCONN) {
126 // connection failed
127 WSASetLastError(so_error);
128 return SOCKET_ERROR;
129 }
130
131 return 0; // success
132 }

134 // try to connect or throw on failure
135 void connect(const std::string &host, int port, int timeout_ms = 0) {
136 if (is_connected()) {
137 close();
138 }
139 struct addrinfo hints {};
140 ZeroMemory(&hints, sizeof(hints));
141
142 hints.ai_family = AF_UNSPEC; // To work with IPv4, IPv6, and so on
143 hints.ai_socktype = SOCK_STREAM; // TCP
144 hints.ai_flags = AI_NUMERICSERV; // port passed as as numeric value
145 hints.ai_protocol = 0;
146
147 timeval tv;
148 tv.tv_sec = timeout_ms / 1000;
149 tv.tv_usec = (timeout_ms % 1000) * 1000;
150
151 auto port_str = std::to_string(port);
152 struct addrinfo *addrinfo_result;
153 auto rv = ::getaddrinfo(host.c_str(), port_str.c_str(), &hints, &addrinfo_result);
154 int last_error = 0;
155 if (rv != 0) {
156 last_error = ::WSAGetLastError();
157 WSACleanup();
158 throw_winsock_error_("getaddrinfo failed", last_error);
159 }
160
161 // Try each address until we successfully connect(2).
162 for (auto *rp = addrinfo_result; rp != nullptr; rp = rp->ai_next) {
163 socket_ = socket(rp->ai_family, rp->ai_socktype, rp->ai_protocol);
164 if (socket_ == INVALID_SOCKET) {
165 last_error = ::WSAGetLastError();
166 WSACleanup();
167 continue;
168 }
169 if (connect_socket_with_timeout(socket_, rp->ai_addr, (int)rp->ai_addrlen, tv) == 0) {
170 last_error = 0;
171 break;
172 }
173 last_error = WSAGetLastError();
174 ::closesocket(socket_);
175 socket_ = INVALID_SOCKET;
176 }
177 ::freeaddrinfo(addrinfo_result);
178 if (socket_ == INVALID_SOCKET) {
179 WSACleanup();
180 throw_winsock_error_("connect failed", last_error);
181 }
182 if (timeout_ms > 0) {
183 DWORD tv = static_cast<DWORD>(timeout_ms);
184 ::setsockopt(socket_, SOL_SOCKET, SO_RCVTIMEO, (const char *)&tv, sizeof(tv));
185 ::setsockopt(socket_, SOL_SOCKET, SO_SNDTIMEO, (const char *)&tv, sizeof(tv));
186 }
187
188 // set TCP_NODELAY
189 int enable_flag = 1;
190 ::setsockopt(socket_, IPPROTO_TCP, TCP_NODELAY, reinterpret_cast<char *>(&enable_flag),
191 sizeof(enable_flag));
192 }

194 // Send exactly n_bytes of the given data.
195 // On error close the connection and throw.
196 void send(const char *data, size_t n_bytes) {
197 size_t bytes_sent = 0;
198 while (bytes_sent < n_bytes) {
199 const int send_flags = 0;
200 auto write_result =
201 ::send(socket_, data + bytes_sent, (int)(n_bytes - bytes_sent), send_flags);
202 if (write_result == SOCKET_ERROR) {
203 int last_error = ::WSAGetLastError();
204 close();
205 throw_winsock_error_("send failed", last_error);
206 }
207
208 if (write_result == 0) // (probably should not happen but in any case..)
209 {
210 break;
211 }
212 bytes_sent += static_cast<size_t>(write_result);
213 }
214 }
