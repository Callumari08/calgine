111template<typename T_behaviour>
112requires std::derived_from<T_behaviour, Behaviour>
113bool GameObject::add_behaviour()
114{
115 auto [it, inserted] = behaviours.emplace(
116 typeid(T_behaviour),
117 nullptr
118 );
119
120 if (!inserted)
121 {
122 Log::get_engine_logger()->warn("Behaviour already exists on GameObject: {}", get_name());
123 return false;
124 }
125
126 std::unique_ptr<Behaviour> owned;
127
128 try
129 {
130 owned.reset(Behaviour::create<T_behaviour>());
131 owned->attach_owner(this);
132 }
133 catch (...) {
134 behaviours.erase(it);
135 throw;
136 }
137
138 it->second = std::move(owned);
139 return true;
140}

144T_behaviour* GameObject::get_behaviour() const
145{
146 auto it = behaviours.find(typeid(T_behaviour));
147 if (it == behaviours.end())
