154

156 Iterator(GameObject* root) {
157 if (root) stack.push(root);
158 }

160 Iterator() {}
161
162 GameObject& operator*() { return *stack.top(); }
163 GameObject* operator->() { return stack.top(); }
164
165 Iterator& operator++() {
166 GameObject* current = stack.top();
167 stack.pop();
168
169 auto& children = current->children;
170 for (auto it = children.rbegin(); it != children.rend(); ++it) {
171 stack.push(it->get());
172 }
173 return *this;
174 }
175
176 bool operator!=(const Iterator& other) const {
177 return stack.size() != other.stack.size();
178 }
179 };
