355template<typename T_behaviour>
356requires std::derived_from<T_behaviour, Behaviour>
357bool GameObject::add_behaviour()
358{
359 auto [it, inserted] = behaviours.emplace(
360 typeid(T_behaviour),
361 nullptr
362 );
363
364 if (!inserted)
365 {
366 Log::get_engine_logger()->warn("Behaviour already exists on GameObject: {}", get_name());
367 return false;
368 }
369
370 std::unique_ptr<Behaviour> owned;
371
372 try
373 {
374 owned.reset(Behaviour::create<T_behaviour>());
375 owned->attach_owner(this);
376 }
377 catch (...) {
378 behaviours.erase(it);
379 throw;
380 }
381
382 it->second = std::move(owned);
383 return true;
384}
