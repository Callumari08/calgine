124 stack.pop();

145requires std::derived_from<T_behaviour, Behaviour>
146T_behaviour* GameObject::add_behaviour(Args&&... args)
147{
148 auto [it, inserted] = behaviours.emplace(
149 typeid(T_behaviour),
150 nullptr
151 );
152 if (!inserted)
153 {
154 Log::get_engine_logger()->warn("Behaviour already exists on GameObject: {}", get_name());
155 return nullptr;
156 }
157 std::unique_ptr<Behaviour> owned;
158
159
160 owned.reset(Behaviour::create<T_behaviour>(std::forward<Args>(args)...));
161
162 if(!owned->attach_owner(this))
163 {
164 behaviours.erase(it);
165 return nullptr;
166 }
167
168 it->second = std::move(owned);
169 return static_cast<T_behaviour*>(it->second.get());
170}
