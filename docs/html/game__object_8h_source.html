109template<typename T_behaviour>
110requires std::derived_from<T_behaviour, Behaviour>
111bool GameObject::add_behaviour()
112{
113 auto [it, inserted] = behaviours.emplace(
114 typeid(T_behaviour),
115 nullptr
116 );
117
118 if (!inserted)
119 {
120 std::cout << "Type: `" << type_name<T_behaviour>() << "` already exists on " << get_name() << "\n";
121 return false;
122 }
123
124 std::unique_ptr<Behaviour> owned;
125
126 try
127 {
128 owned.reset(Behaviour::create<T_behaviour>());
129 owned->attach_owner(this);
130 }
131 catch (...) {
132 behaviours.erase(it);
133 throw;
134 }
135
136 it->second = std::move(owned);
137 return true;
138}
