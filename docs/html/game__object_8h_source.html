104template<typename T_behaviour>
105requires std::derived_from<T_behaviour, Behaviour>
106bool GameObject::add_behaviour()
107{
108 auto [it, inserted] = behaviours.emplace(
109 typeid(T_behaviour),
110 nullptr
111 );
112
113 if (!inserted)
114 {
115 std::cout << "Type: `" << type_name<T_behaviour>() << "` already exists on " << get_name() << "\n";
116 return false;
117 }
118
119 std::unique_ptr<Behaviour> owned;
120
121 try
122 {
123 owned.reset(Behaviour::create<T_behaviour>());
124 owned->attach_owner(this);
125 }
126 catch (...) {
127 behaviours.erase(it);
128 throw;
129 }
130
131 it->second = std::move(owned);
132 return true;
133}
